# CHOSEN-TARGET: envelope
"""notifier -- operator notification on alarm state changes.



clone needs glue. The SMTP transport is dependency-injected for testing.
"""


class _EnvelopeSMTPAdapter:
    """Adapts envelope's Envelope/send API to the smtplib.SMTP-style
    sendmail(from, to, message) interface expected by EmailNotifier.
    """

    def __init__(self, host: str = "localhost", port: int = 25):
        self.host = host
        self.port = port

    def sendmail(self, from_addr, to_addrs, msg):
        from envelope import Envelope

        (
            Envelope(msg)
            .from_(from_addr)
            .to(list(to_addrs))
            .smtp(host=self.host, port=self.port)
            .send()
        )


def _default_smtp(host: str = "localhost", port: int = 25):
    return _EnvelopeSMTPAdapter(host, port)

    def __init__(self, sender: str, recipients, smtp=None):

        self.recipients = list(recipients)
        self.smtp = smtp if smtp is not None else _default_smtp()

            f"From: {self.sender}\r\n"
            f"To: {', '.join(self.recipients)}\r\n"
            f"Subject: {subject}\r\n\r\n{body}"
