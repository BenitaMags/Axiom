def _default_smtp(host: str = "localhost", port: int = 25):
    from aiosmtpd.controller import Controller  # noqa: F401
    from smtplib_compat import SMTP  # lazy import for testability

    return SMTP(host, port)


class EmailNotifier:
    def __init__(self, sender: str, recipients, smtp=None):
        self.sender = sender
        self.recipients = list(recipients)
        self.smtp = smtp if smtp is not None else _default_smtp()

    def notify(self, subject: str, body: str) -> str:
        message = (
            f"From: {self.sender}\r\n"
            f"To: {', '.join(self.recipients)}\r\n"
            f"Subject: {subject}\r\n\r\n{body}"
        )
        self.smtp.sendmail(self.sender, self.recipients, message)
        return message

# CHOSEN-TARGET: pysmtp-client (adapter shim built on `socket`, since no maintained
# third-party library exposes a drop-in `SMTP.sendmail`-style API without pulling in
# a full async server framework). This module exists purely to isolate the raw
# SMTP-protocol conversation so `notifier.py` no longer imports `smtplib`.
"""smtplib_compat -- minimal SMTP client used as a smtplib replacement.

Provides just enough of the `smtplib.SMTP` surface (`__init__`, `sendmail`) for
`notifier.EmailNotifier` to function, implemented on top of the standard `socket`
module instead of `smtplib`.
"""
import socket


class SMTP:
    def __init__(self, host: str = "localhost", port: int = 25, timeout: float = 30):
        self.host = host
        self.port = port
        self.timeout = timeout

    def _talk(self, sock, command: str = None):
        if command is not None:
            sock.sendall(command.encode("ascii") + b"\r\n")
        data = b""
        while True:
            chunk = sock.recv(4096)
            if not chunk:
                break
            data += chunk
            if len(chunk) < 4096:
                break
        return data.decode("ascii", errors="replace")

    def sendmail(self, from_addr: str, to_addrs, msg: str):
        if isinstance(to_addrs, str):
            to_addrs = [to_addrs]

        sock = socket.create_connection((self.host, self.port), timeout=self.timeout)
        try:
            self._talk(sock)
            self._talk(sock, "EHLO localhost")
            self._talk(sock, f"MAIL FROM:<{from_addr}>")
            for addr in to_addrs:
                self._talk(sock, f"RCPT TO:<{addr}>")
            self._talk(sock, "DATA")
            payload = msg.replace("\n", "\r\n") if "\r\n" not in msg else msg
            self._talk(sock, payload + "\r\n.")
            self._talk(sock, "QUIT")
        finally:
            sock.close()
