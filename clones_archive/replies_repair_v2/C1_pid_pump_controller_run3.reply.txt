"""controller -- PID pump controller regulating tank level to a setpoint.


Because the substitution *removes* a third-party behavior, the PID computation becomes
first-party code that the LMA must re-implement and diversify -- hence Category C. The
surrounding logic (setpoint ramping, output clamping, AUTO/MANUAL mode) is already
first-party. The PID object is dependency-injected for testing.
"""
import time

from io_map import speed_to_counts


class _SimplePID:
    """A minimal, self-contained PID controller.

    Mirrors the subset of ``simple_pid.PID`` behavior used by this module:
    a callable object holding ``kp``, ``ki``, ``kd``, a settable ``setpoint``
    attribute, and ``output_limits`` used to clamp both the computed output
    and (to avoid integral windup) the accumulated integral term.
    """

    def __init__(self, kp, ki, kd, setpoint=0.0, output_limits=(None, None)):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.setpoint = setpoint
        self.output_limits = output_limits

        self._integral = 0.0
        self._last_input = None
        self._last_time = None

    def _clamp(self, value, limits):
        lower, upper = limits
        if value is None:
            return None
        if upper is not None and value > upper:
            value = upper
        if lower is not None and value < lower:
            value = lower
        return value

    def __call__(self, input_, dt=None):
        now = time.monotonic()

        if dt is None:
            if self._last_time is None:
                dt = 1e-16
            else:
                dt = now - self._last_time
                if dt <= 0.0:
                    dt = 1e-16

        error = self.setpoint - input_

        proportional = self.kp * error

        self._integral += self.ki * error * dt
        self._integral = self._clamp(self._integral, self.output_limits)

        if self._last_input is None:
            d_input = 0.0
        else:
            d_input = input_ - self._last_input
        derivative = -self.kd * d_input / dt

        output = proportional + self._integral + derivative
        output = self._clamp(output, self.output_limits)

        self._last_input = input_
        self._last_time = now

        return output


def _make_pid(kp, ki, kd, setpoint):
    return _SimplePID(kp, ki, kd, setpoint=setpoint, output_limits=(0.0, 100.0))


class PumpController:
    def __init__(self, setpoint, kp=2.0, ki=0.5, kd=0.0, ramp=0.5, pid=None):
        self.target = setpoint
        self.applied_setpoint = 0.0
        self.ramp = ramp
        self.mode = "AUTO"
        self.pid = pid if pid is not None else _make_pid(kp, ki, kd, 0.0)

    def _ramp_setpoint(self):
        """First-party logic: move the applied setpoint toward the target by <= ramp."""
        if self.applied_setpoint < self.target:
            self.applied_setpoint = min(self.target, self.applied_setpoint + self.ramp)
        elif self.applied_setpoint > self.target:
            self.applied_setpoint = max(self.target, self.applied_setpoint - self.ramp)
        self.pid.setpoint = self.applied_setpoint

    def update(self, level) -> int:
        """Compute the pump-speed command (in DAC counts) for the measured level."""
        if self.mode == "MANUAL":
            return speed_to_counts(0.0)
        self._ramp_setpoint()
        speed = self.pid(level)
        speed = max(0.0, min(100.0, speed))  # clamp before actuation
        return speed_to_counts(speed)

"""io_map -- field I/O scaling for the pump controller (first-party logic)."""


def counts_to_level(counts, adc_span=(0, 4095), level_range=(0.0, 10.0)) -> float:
    """Convert raw level-sensor ADC counts to meters."""
    lo, hi = adc_span
    a, b = level_range
    return a + (counts - lo) * (b - a) / (hi - lo)


def speed_to_counts(speed_pct, dac_span=(0, 4095)) -> int:
    """Convert a 0-100% pump-speed command to DAC counts (clamped)."""
    lo, hi = dac_span
    speed_pct = max(0.0, min(100.0, speed_pct))
    return int(round(lo + speed_pct * (hi - lo) / 100.0))

"""plant -- a minimal tank simulator used to exercise the controller in closed loop.

This is test/reference scaffolding (a first-order integrator), not part of the shipped
controller, but it provides the reference trajectory the equivalence oracle checks.
"""


class Tank:
    def __init__(self, level=0.0, inflow_gain=0.05, outflow=0.02, area=1.0):
        self.level = level
        self.inflow_gain = inflow_gain
        self.outflow = outflow
        self.area = area

    def step(self, pump_speed_pct, dt=1.0) -> float:
        inflow = self.inflow_gain * pump_speed_pct
        self.level += (inflow - self.outflow) * dt / self.area
        self.level = max(0.0, self.level)
        return self.level
