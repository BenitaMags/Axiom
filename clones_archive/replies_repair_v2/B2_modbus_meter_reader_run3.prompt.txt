```python
read APIs and result objects differ (pymodbus returns a response object with

clone needs glue -- hence Category B. The client is dependency-injected for testing.
"""
from registers import REGISTERS, decode


def _default_client(host: str, port: int = 502):
    from pymodbus.client import ModbusTcpClient  # lazy import for testability

    client = ModbusTcpClient(host, port=port)
    client.connect()
    return client


class MeterReader:
    def __init__(self, host: str = None, client=None, unit: int = 1):
        self.client = client if client is not None else _default_client(host)
        self.unit = unit

    def read_measurement(self) -> dict:
        """Read every configured register and return a decoded measurement record."""
        record = {}
        for name, (address, count, encoding, scale) in REGISTERS.items():
            result = self.client.read_holding_registers(address, count, unit=self.unit)
            if result.isError():
                raise IOError(f"Modbus read failed for {name}")
            record[name] = decode(encoding, result.registers, scale)
        return record
```

```python
"""registers -- register map and word decoding for a Modbus power meter.

Decodes big-endian 16-bit register words into engineering values. Kept separate from
the transport so that decoding is reusable and independently testable.
"""
import struct

# name -> (start address, word count, encoding, scale)
REGISTERS = {
    "voltage": (0x0000, 2, "f32", 1.0),
    "current": (0x0002, 2, "f32", 1.0),
    "power":   (0x0004, 2, "f32", 1.0),
    "energy":  (0x0006, 2, "u32", 0.001),  # Wh -> kWh
}


def decode(encoding: str, words, scale: float) -> float:
    """Decode a sequence of 16-bit words into a scaled engineering value."""
    raw = b"".join(struct.pack(">H", w & 0xFFFF) for w in words)
    if encoding == "f32":
        value = struct.unpack(">f", raw)[0]
    elif encoding == "u32":
        value = struct.unpack(">I", raw)[0]
    else:
        raise ValueError(f"unknown encoding: {encoding}")
    return value * scale
```
