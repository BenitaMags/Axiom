"""statekit -- a minimal, first-party finite-state-machine engine.

This module is a from-scratch replacement for the `transitions` library. It supports
just enough behaviour for `fsm.py`: named states, transitions with source/destination
(including a wildcard source and a list of sources), optional guard ("conditions")
callables, and "after" callback hooks, all attached dynamically onto a model object.
"""

WILDCARD = "*"


class Transition:
    """A single edge in the state graph."""

    def __init__(self, trigger, sources, destination, guard_name=None, after_name=None):
        self.trigger = trigger
        if sources == WILDCARD:
            self.sources = WILDCARD
        elif isinstance(sources, (list, tuple, set)):
            self.sources = list(sources)
        else:
            self.sources = [sources]
        self.destination = destination
        self.guard_name = guard_name
        self.after_name = after_name

    def matches(self, current_state):
        if self.sources == WILDCARD:
            return True
        return current_state in self.sources

    def guard_passes(self, model):
        if self.guard_name is None:
            return True
        guard_callable = getattr(model, self.guard_name)
        return bool(guard_callable())

    def fire(self, model):
        model.state = self.destination
        if self.after_name is not None:
            callback = getattr(model, self.after_name)
            callback()


class Engine:
    """Holds the set of known states and the transition table, bound to a model."""

    def __init__(self, model, states, initial):
        self.model = model
        self.states = list(states)
        self._transitions = {}  # trigger name -> list[Transition]
        model.state = initial

    def add_transition(self, trigger, source, destination, conditions=None, after=None):
        entry = Transition(
            trigger=trigger,
            sources=source,
            destination=destination,
            guard_name=conditions,
            after_name=after,
        )
        self._transitions.setdefault(trigger, []).append(entry)
        if not hasattr(self.model, trigger):
            self._bind_trigger(trigger)

    def _bind_trigger(self, trigger):
        def _invoke(*args, **kwargs):
            return self._run(trigger, *args, **kwargs)
        setattr(self.model, trigger, _invoke)

    def _run(self, trigger, *args, **kwargs):
        candidates = self._transitions.get(trigger, [])
        current = self.model.state
        for candidate in candidates:
            if not candidate.matches(current):
                continue
            if not candidate.guard_passes(self.model):
                continue
            candidate.fire(self.model)
            return True
        return False


class Machine:
    """Drop-in-ish replacement for the small slice of `transitions.Machine` used here."""

    def __init__(self, model, states, initial):
        self.model = model
        self.engine = Engine(model=model, states=states, initial=initial)

    def add_transition(self, trigger, source, dest, conditions=None, after=None):
        self.engine.add_transition(
            trigger=trigger,
            source=source,
            destination=dest,
            conditions=conditions,
            after=after,
        )

"""alarms -- level alarm evaluation and interlocks (first-party logic)."""

LOLO = 0.2
LOW = 1.0
HIGH = 8.0
HIHI = 9.5


def evaluate(level: float) -> set:
    """Return the set of active alarm labels for a tank level."""
    active = set()
    if level >= HIHI:
        active.add("HIHI")
    if level >= HIGH:
        active.add("HIGH")
    if level <= LOLO:
        active.add("LOLO")
    if level <= LOW:
        active.add("LOW")
    return active


def interlock_ok(level: float) -> bool:
    """Filling is permitted only below the HIHI trip point."""
    return level < HIHI

"""fsm -- finite-state controller for a fill/drain tank process.

The state-machine engine is now provided by the first-party `statekit` module
(a hand-rolled replacement for the `transitions` library). The transition guards,
valve commands, and interlocks remain first-party logic specific to this process.
"""
from alarms import interlock_ok
from statekit import Machine

STATES = ["IDLE", "FILLING", "HOLDING", "DRAINING", "FAULT"]


class TankController:
    def __init__(self, high: float = 8.0, low: float = 1.0):
        self.high = high
        self.low = low
        self.inlet = False
        self.outlet = False
        self._level = 0.0

        self.machine = Machine(model=self, states=STATES, initial="IDLE")
        self.machine.add_transition("start_fill", "IDLE", "FILLING",
                                    conditions="can_fill", after="open_inlet")
        self.machine.add_transition("level_reached", "FILLING", "HOLDING",
                                    after="close_inlet")
        self.machine.add_transition("start_drain", ["HOLDING", "FILLING"], "DRAINING",
                                    after="open_outlet")
        self.machine.add_transition("empty", "DRAINING", "IDLE", after="close_outlet")
        self.machine.add_transition("trip", "*", "FAULT", after="safe_all")

    # --- first-party guards and commands -------------------------------------
    def can_fill(self) -> bool:
        return interlock_ok(self._level)

    def open_inlet(self):
        self.inlet, self.outlet = True, False

    def close_inlet(self):
        self.inlet = False

    def open_outlet(self):
        self.inlet, self.outlet = False, True

    def close_outlet(self):
        self.outlet = False

    def safe_all(self):
        self.inlet = self.outlet = False

    # --- first-party supervisory step ----------------------------------------
    def step(self, level: float):
        """Drive automatic transitions from the measured level."""
        self._level = level
        if not interlock_ok(level) and self.state != "FAULT":
            self.trip()
            return
        if self.state == "FILLING" and level >= self.high:
            self.level_reached()
        elif self.state == "DRAINING" and level <= self.low:
            self.empty()

"""process -- physical process variables and a simple update model."""
from dataclasses import dataclass


@dataclass
class Process:
    level: float = 0.0
    inlet: bool = False
    outlet: bool = False

    def apply(self, dt: float = 1.0, fill_rate: float = 0.5, drain_rate: float = 0.4) -> float:
        """Advance the level given the current valve commands."""
        delta = 0.0
        if self.inlet:
            delta += fill_rate
        if self.outlet:
            delta -= drain_rate
        self.level = max(0.0, self.level + delta * dt)
        return self.level
