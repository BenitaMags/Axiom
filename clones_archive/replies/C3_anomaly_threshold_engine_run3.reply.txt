```python
"""engine -- streaming out-of-band anomaly detector with hysteresis.

Ties together per-tag hysteresis detection (first-party), configuration, and operator
notification. Emits one event per alarm state change and notifies on each transition.
"""
from hysteresis import Hysteresis


class AnomalyEngine:
    def __init__(self, config: dict, notifier=None):
        built = {}
        for tag, cfg in config.items():
            built[tag] = Hysteresis(cfg.high, cfg.low)
        self.detectors = built
        self.notifier = notifier
        self.events = []

    def process(self, reading: dict):
        """Process one {tag, value} reading; return an event on state change else None."""
        tag_name = reading["tag"]
        matched_detector = self.detectors.get(tag_name)
        if matched_detector is None:
            return None

        prior_state = matched_detector.state
        new_state = matched_detector.update(reading["value"])

        if new_state == prior_state:
            return None

        emitted_event = {"tag": tag_name, "state": new_state, "value": reading["value"]}
        self.events.append(emitted_event)

        if self.notifier is not None:
            subject_line = f"{tag_name} {new_state}"
            self.notifier.notify(subject_line, str(emitted_event))

        return emitted_event

    def process_stream(self, readings) -> list:
        """Process an iterable of readings, returning only the emitted events."""
        results = []
        for single_reading in readings:
            outcome = self.process(single_reading)
            if outcome is not None:
                results.append(outcome)
        return results
```

```python
"""engine_config -- per-tag threshold configuration for the anomaly engine."""
from dataclasses import dataclass


@dataclass
class TagConfig:
    tag: str
    high: float
    low: float


def load(config: dict) -> dict:
    """Build {tag: TagConfig} from a config dict of the form {"tags": [ ... ]}."""
    output = {}
    entries = config["tags"]
    idx = 0
    while idx < len(entries):
        entry = entries[idx]
        candidate = TagConfig(tag=entry["tag"], high=entry["high"], low=entry["low"])
        if not (candidate.low < candidate.high):
            raise ValueError(f"tag {candidate.tag}: low must be < high")
        output[candidate.tag] = candidate
        idx += 1
    return output
```

```python
"""hysteresis -- dual-threshold hysteresis detector (first-party core logic).

This is the detection logic the LMA must rewrite into an equivalent but structurally
different form. A value at or above `high` trips ALARM; the detector only clears once
the value falls back to or below `low`, preventing chatter within the deadband.
"""


class Hysteresis:
    def __init__(self, high: float, low: float):
        if not (low < high):
            raise ValueError("low must be < high")
        self.high = high
        self.low = low
        self.state = "NORMAL"

    def update(self, value: float) -> str:
        if self.state == "NORMAL":
            if value >= self.high:
                self.state = "ALARM"
        else:
            if value <= self.low:
                self.state = "NORMAL"
        return self.state
```

```python
"""notifier -- operator notification on alarm state changes.

The notification backend is the substitution target: `smtplib` (stdlib SMTP) ->
`apprise` (multi-service notifications). The API differs (sendmail vs. notify), so a
clone needs glue: an `AppriseSmtpAdapter` exposes a `sendmail`-shaped method backed by
an internal `apprise.Apprise` instance, letting `EmailNotifier` stay structurally
similar to code written against `smtplib` while a real dependency-injected test double
can still stand in for `smtp` exactly as before.
"""


def _build_apprise_target(host: str = "localhost", port: int = 25):
    """Construct a default apprise-backed transport pointed at a mailto:// endpoint."""
    import apprise  # lazy import for testability

    apprise_app = apprise.Apprise()
    mailto_url = f"mailto://{host}:{port}"
    apprise_app.add(mailto_url)
    return AppriseSmtpAdapter(apprise_app)


class AppriseSmtpAdapter:
    """Adapter giving an apprise.Apprise instance an smtplib-like `sendmail` method."""

    def __init__(self, apprise_app):
        self._app = apprise_app

    def sendmail(self, sender: str, recipients, message: str):
        # apprise doesn't model sender/recipients the way SMTP does; we fold that
        # context into the notification body so behavior/logging stays equivalent.
        combined_body = f"From: {sender}\r\nTo: {', '.join(recipients)}\r\n\r\n{message}"
        self._app.notify(body=combined_body)
        return None


def _default_smtp(host: str = "localhost", port: int = 25):
    return _build_apprise_target(host, port)


class EmailNotifier:
    def __init__(self, sender: str, recipients, smtp=None):
        self.sender = sender
        self.recipients = list(recipients)
        self.smtp = smtp if smtp is not None else _default_smtp()

    def notify(self, subject: str, body: str) -> str:
        header_from = f"From: {self.sender}\r\n"
        header_to = f"To: {', '.join(self.recipients)}\r\n"
        header_subject = f"Subject: {subject}\r\n\r\n"
        message = header_from + header_to + header_subject + body
        self.smtp.sendmail(self.sender, self.recipients, message)
        return message
```
