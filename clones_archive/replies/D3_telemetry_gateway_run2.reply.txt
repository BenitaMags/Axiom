"""estimator -- first-party engineering-unit estimation with anomaly flagging.

This is the core application logic the LMA must rewrite into an equivalent but
structurally different form. It scales raw counts to engineering units, smooths with a
moving average, and flags a rate-of-change anomaly when consecutive samples jump too far.
"""


class Estimator:
    def __init__(self, scale: float = 0.1, window: int = 5, roc_limit: float = 5.0):
        self.scale = scale
        self.window = window
        self.roc_limit = roc_limit
        self.history = []
        self.previous_value = None

    def update(self, raw_counts: float) -> dict:
        engineering_value = raw_counts * self.scale

        self.history.append(engineering_value)
        overflow = len(self.history) - self.window
        while overflow > 0:
            del self.history[0]
            overflow -= 1

        total = 0.0
        for sample in self.history:
            total += sample
        moving_average = total / len(self.history)

        if self.previous_value is None:
            delta = 0.0
        else:
            delta = engineering_value - self.previous_value
            if delta < 0:
                delta = -delta

        self.previous_value = engineering_value

        is_anomalous = delta > self.roc_limit

        return {
            "value": engineering_value,
            "smoothed": moving_average,
            "anomaly": is_anomalous,
        }

"""gateway -- telemetry gateway: read raw counts, estimate, publish.

Combines all three transformations in one program:
  * LSA + GSA: swap the MQTT client (`paho-mqtt` -> `aiomqtt`) and write async glue.
  * LMA: rewrite the first-party estimation/anomaly logic in `estimator.py`.

`reader` is an injected callable `tag -> raw_counts` (the field acquisition source).
"""
from estimator import Estimator
from sink import MqttSink


class TelemetryGateway:
    def __init__(self, reader, sink: MqttSink, estimators: dict = None,
                 scale: float = 0.1, window: int = 5, roc_limit: float = 5.0):
        self.reader = reader
        self.sink = sink
        self.estimators = estimators if estimators is not None else {}
        self._cfg = {"scale": scale, "window": window, "roc_limit": roc_limit}

    def _estimator(self, tag: str) -> Estimator:
        found = self.estimators.get(tag)
        if found is None:
            found = Estimator(**self._cfg)
            self.estimators[tag] = found
        return found

    def process(self, tag: str) -> dict:
        """Acquire one sample for *tag*, estimate, publish, and return the record."""
        raw_sample = self.reader(tag)
        estimator = self._estimator(tag)
        record = estimator.update(raw_sample)
        record["tag"] = tag
        self.sink.publish(tag, record)
        return record

    def process_all(self, tags) -> list:
        results = []
        for single_tag in tags:
            results.append(self.process(single_tag))
        return results

"""sink -- MQTT publication of estimated telemetry records.

The MQTT client is one substitution target: `paho-mqtt` (synchronous, callback-based)
-> `aiomqtt` (async, context-managed) -> glue required. The client is injected.

Since the public API of `MqttSink.publish` is synchronous (`tag, record -> int`), we
provide a small adapter that drives an underlying `aiomqtt.Client` (async) from
synchronous code by running a private event loop. If a client is injected (e.g. in
tests), it is used as-is via a uniform `publish(topic, payload) -> rc`-like interface.
"""
import asyncio
import json


class _AiomqttSyncAdapter:
    """Adapts an async `aiomqtt.Client` to the synchronous publish(topic, payload) API."""

    def __init__(self, host: str = "localhost", port: int = 1883):
        self.host = host
        self.port = port
        self._loop = asyncio.new_event_loop()

    async def _do_publish(self, topic: str, payload: bytes):
        import aiomqtt

        async with aiomqtt.Client(hostname=self.host, port=self.port) as client:
            await client.publish(topic, payload=payload)
        return 0

    def publish(self, topic: str, payload: bytes):
        rc = self._loop.run_until_complete(self._do_publish(topic, payload))
        return _PublishResult(rc)

    def __del__(self):
        try:
            if not self._loop.is_closed():
                self._loop.close()
        except Exception:
            pass


class _PublishResult:
    def __init__(self, rc: int):
        self.rc = rc


def _default_client(host: str = "localhost", port: int = 1883):
    return _AiomqttSyncAdapter(host, port)


class MqttSink:
    def __init__(self, topic_base: str = "plant/est", client=None):
        self.topic_base = topic_base
        self.client = client if client is not None else _default_client()

    def publish(self, tag: str, record: dict) -> int:
        payload = json.dumps(record, separators=(",", ":")).encode("utf-8")
        full_topic = "{}/{}".format(self.topic_base, tag)
        info = self.client.publish(full_topic, payload)
        return getattr(info, "rc", 0)
