"""notifier -- operator notification on alarm state changes.

The notification backend is the substitution target: `smtplib` (stdlib SMTP) ->
`apprise` (multi-service notifications). The API differs (sendmail vs. notify), so a
clone needs glue that builds an apprise `mailto://` URL from sender/recipients and
formats the message body itself, then hands the payload to an injected apprise-like
transport object exposing `notify(...)`. The transport is dependency-injected for
testing, mirroring the original SMTP injection point.
"""


def _build_mailto_url(sender: str, recipients) -> str:
    """Construct an apprise-compatible mailto:// URL encoding sender & recipients."""
    joined_targets = ",".join(recipients)
    return f"mailto://{sender}/{joined_targets}"


def _default_apprise(url: str):
    import apprise  # lazy import for testability

    transport = apprise.Apprise()
    transport.add(url)
    return transport


class EmailNotifier:
    def __init__(self, sender: str, recipients, smtp=None):
        self.sender = sender
        self.recipients = list(recipients)
        self._mailto_url = _build_mailto_url(self.sender, self.recipients)
        self.smtp = smtp if smtp is not None else _default_apprise(self._mailto_url)

    def _render_message(self, subject: str, body: str) -> str:
        header_lines = [
            f"From: {self.sender}",
            f"To: {', '.join(self.recipients)}",
            f"Subject: {subject}",
            "",
        ]
        return "\r\n".join(header_lines) + body

    def notify(self, subject: str, body: str) -> str:
        rendered = self._render_message(subject, body)
        self.smtp.notify(title=subject, body=rendered)
        return rendered

"""engine -- streaming out-of-band anomaly detector with hysteresis.

Ties together per-tag hysteresis detection (first-party), configuration, and operator
notification. Emits one event per alarm state change and notifies on each transition.
"""
from hysteresis import Hysteresis


class AnomalyEngine:
    def __init__(self, config: dict, notifier=None):
        built = {}
        for tag_name, tag_cfg in config.items():
            built[tag_name] = Hysteresis(tag_cfg.high, tag_cfg.low)
        self.detectors = built
        self.notifier = notifier
        self.events = []

    def _lookup(self, tag):
        return self.detectors.get(tag)

    def process(self, reading: dict):
        """Process one {tag, value} reading; return an event on state change else None."""
        tag_name = reading["tag"]
        detector = self._lookup(tag_name)
        if detector is None:
            return None

        state_before = detector.state
        state_after = detector.update(reading["value"])

        if state_after == state_before:
            return None

        new_event = {"tag": tag_name, "state": state_after, "value": reading["value"]}
        self.events.append(new_event)

        if self.notifier is not None:
            subject = f"{tag_name} {state_after}"
            self.notifier.notify(subject, str(new_event))

        return new_event

    def process_stream(self, readings) -> list:
        """Process an iterable of readings, returning only the emitted events."""
        results = []
        for item in readings:
            outcome = self.process(item)
            if outcome is not None:
                results.append(outcome)
        return results

"""engine_config -- per-tag threshold configuration for the anomaly engine."""
from dataclasses import dataclass


@dataclass
class TagConfig:
    tag: str
    high: float
    low: float


def load(config: dict) -> dict:
    """Build {tag: TagConfig} from a config dict of the form {"tags": [ ... ]}."""
    tags_by_name = {}
    for entry in config["tags"]:
        current = TagConfig(tag=entry["tag"], high=entry["high"], low=entry["low"])
        if not (current.low < current.high):
            raise ValueError(f"tag {current.tag}: low must be < high")
        tags_by_name[current.tag] = current
    return tags_by_name

"""hysteresis -- dual-threshold hysteresis detector (first-party core logic).

This is the detection logic the LMA must rewrite into an equivalent but structurally
different form. A value at or above `high` trips ALARM; the detector only clears once
the value falls back to or below `low`, preventing chatter within the deadband.
"""

NORMAL_STATE = "NORMAL"
ALARM_STATE = "ALARM"


class Hysteresis:
    def __init__(self, high: float, low: float):
        if not (low < high):
            raise ValueError("low must be < high")
        self.high = high
        self.low = low
        self.state = NORMAL_STATE

    def update(self, value: float) -> str:
        if self.state == NORMAL_STATE:
            if value >= self.high:
                self.state = ALARM_STATE
        else:
            if value <= self.low:
                self.state = NORMAL_STATE
        return self.state
