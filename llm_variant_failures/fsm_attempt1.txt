```python
"""Generic trigger-table driven finite-state machine engine.

This is a structurally different re-implementation of the minimal
`transitions.Machine` substitute previously embedded in fsm.py. It works by
building a table of Transition specifications, then exposing one trigger
callable per unique trigger name.
"""
from typing import Callable, Iterable, List, NamedTuple, Optional, Sequence, Union


class MachineError(Exception):
    """Raised when a triggered transition has no matching source/condition."""


def _as_tuple(value: Optional[Union[str, Iterable[str]]]) -> tuple:
    if value is None:
        return ()
    if isinstance(value, str):
        return (value,)
    return tuple(value)


class Transition(NamedTuple):
    trigger: str
    sources: tuple
    dest: str
    conditions: tuple = ()
    after: tuple = ()


class Engine:
    """Minimal stand-in for transitions.Machine covering the API used here."""

    def __init__(self, model, states, initial):
        self.model = model
        self.states = list(states)
        self._transitions: List[Transition] = []
        model.state = initial

    def add_transition(self, trigger, source, dest, conditions=None, after=None):
        sources = _as_tuple(source)
        conditions_tuple = _as_tuple(conditions)
        after_tuple = _as_tuple(after)

        spec = Transition(trigger, sources, dest, conditions_tuple, after_tuple)
        self._transitions.append(spec)
        setattr(self.model, trigger, self._make_trigger(spec))

    def _make_trigger(self, spec: Transition) -> Callable[..., bool]:
        model = self.model

        def trigger_fn(*args, **kwargs):
            current = model.state
            if spec.sources != ("*",) and current not in spec.sources:
                raise MachineError(
                    "Can't trigger event %s from state %s!" % (spec.trigger, current)
                )

            for cond_name in spec.conditions:
                if not getattr(model, cond_name)():
                    return False

            model.state = spec.dest

            for after_name in spec.after:
                getattr(model, after_name)()

            return True

        return trigger_fn
```
