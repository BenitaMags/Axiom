```python
"""scaling -- numeric conversions between physical units and raw Modbus register counts."""

from typing import NamedTuple


class LinearScale(NamedTuple):
    """A simple multiplicative unit conversion: raw register value -> physical unit."""

    factor: float

    def to_physical(self, raw: int) -> float:
        return raw / self.factor


class PercentToCounts(NamedTuple):
    """Maps a 0-100% span onto a 0-4095 raw register span, clamping out-of-range input."""

    counts_per_percent: float
    low: float = 0.0
    high: float = 100.0

    def to_counts(self, percent: float) -> int:
        clamped = max(self.low, min(self.high, percent))
        return int(round(clamped * self.counts_per_percent))
```
